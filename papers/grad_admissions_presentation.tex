\begin{frame}[label=m]
\frametitle{Motivation}
\begin{itemize}
\item \textbf{Economic Problem:}\ Selecting best options under costly evaluation
\vspace{5pt}
\begin{itemize}
\item Determining each option's true quality costs decision-maker time and effort
\vspace{5pt}
\item Low-cost screening of all options $\Rightarrow$ high-cost evaluation of surviving options
\vspace{5pt}
\item \textbf{Examples:}\ Admissions and hiring, clinical trials, matching platforms
\vspace{10pt}
\end{itemize}
\item \textbf{Why Graduate Admissions:}\ Benefits graduate programs and future applicants
\vspace{5pt}
\begin{itemize}
\item Graduate programs invest time and money into students' development
\vspace{5pt}
\item If programs were undervaluing applicants with certain characteristics, \\ future such applicants may receive formerly-withheld opportunities
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[label=rq]
\frametitle{Research Questions}
\begin{enumerate}
\item Which factors in applications affect admission and success in graduate school?
\vspace{5pt}
\begin{itemize}
\item Literature measures success with program completion and job placement ranking
\vspace{10pt}
\end{itemize}
\item Are admissions committees optimizing, or are they making mistakes?
\vspace{5pt}
\begin{itemize}
\item Goal is to accept applicants with highest success probabilities who will matriculate
\vspace{5pt}
\item Predictors of admission only (success only) may be overvalued (undervalued)
\vspace{10pt}
\end{itemize}
\item If they are making mistakes, how can they make better admissions decisions?
\vspace{5pt}
\begin{itemize}
\item In turn, by changing decision rules, how much better of a cohort can they admit?
\vspace{10pt}
\end{itemize}
\item[$\bigstar$] This paper's methods generalize to admissions, hiring, and other settings
\end{enumerate}
\end{frame}

\begin{frame}[label=mmr]
\frametitle{Methodology and Main Results}
\begin{itemize}
\item Build models of admissions committee's decision problem that support:
\vspace{5pt}
\begin{itemize}
\item Statistical tests of whether committee's decision rule aligns with desired objective
\vspace{5pt}
\item Counterfactual simulations quantifying gains from deviating to model's decision rule
\vspace{10pt}
\end{itemize}
\item Estimate models with admissions data from a major research university
\vspace{5pt}
\begin{itemize}
\item Do estimated success probabilities align with those implied by committee's decisions?\!\!\!\!
\vspace{5pt}
\item Tests reject optimality and counterfactual deviation gains statistically significant when committee misvalues applicant characteristics, but not when it optimizes
\end{itemize}
\end{itemize}
\end{frame}

\section{Structural Models}
\addtocounter{framenumber}{-1}
\begin{frame}[label=wma]
\frametitle{Why Model Admissions?}
\begin{itemize}
\item \textbf{Reason 1:}\ Test whether committee's decision rule aligns with desired objective
\vspace{10pt}
\begin{itemize}
\item Per literature, committee should accept applicants with highest success probabilities
\vspace{10pt}
\item Committees also accept applicants with higher matriculation probabilities
\vspace{20pt}
\end{itemize}
\item \textbf{Reason 2:}\ Find counterfactual decision rule in which committee optimizes
\vspace{10pt}
\begin{itemize}
\item Quantify gains from deviating from actual to counterfactual decision rule
\end{itemize}
\end{itemize}
\end{frame}

\begin{frame}[label=mf]
\frametitle{Model Framework}
\begin{itemize}
\item \textbf{Portfolio Choice Problem:}\ Accept subset of applicants with highest sum of success probabilities subject to capacity constraint on number of acceptances
\only<1>{
\vspace{10pt}
\item For each applicant $n\in\{1,\dots,N\}$, make decision $d_n\in\{\text{Accept},\text{Reject}\}$ given:
}
\end{itemize}
\begin{equation*}
\max_{(d_1,\dots,d_N)\in\{A,R\}^N} \sum_{n=1}^N\underbrace{P(y_n=S|x_n,z_n)}_{\text{Success probability}}\underbrace{\mathds1(d_n=A)}_{\text{Acceptance}}, \: \text{such that:} \: \sum_{n=1}^N\underbrace{\mathds1(d_n=A)}_{\text{Acceptance}}\leq N_A\only<1>{,}
\end{equation*}
\only<1>{\noindent
\begin{itemize}
\item[] where $x=$ objective variables, $z=$ subjective variables, $y\in\{\text{Success},\text{Failure}\}$, $N=$ number of applicants, and $N_A=$ maximum number of acceptances
\vspace{10pt}
\end{itemize}
}
\only<2->{\noindent
\begin{itemize}
\item BEMS \citeyearp{bai2022} prove that portfolio choice problem reduces to cutoff rule problem
\vspace{5pt}
\begin{itemize}
\item Accept applicant if and only if their success probability exceeds cutoff probability
\vspace{5pt}
\item Decisions across applicants now independent, so no need to consider all subsets
\vspace{10pt}
\end{itemize}
\end{itemize}
\hyperlink{pctcr}{\beamerskipbutton{Portfolio Choice to Cutoff Rule}}
}
\end{frame}

\begin{frame}[label=crp]
\frametitle{Cutoff Rule Problem}
\begin{itemize}
\item Start with two-round cutoff rule problem that accounts for costly evaluation:
\vspace{5pt}
\begin{itemize}
\item \textbf{Round 1:}\ See only objective variables (e.g.\ GRE). Can transition application file to Round 2 to see subjective variables (e.g.\ recommendation letters), or can reject
\vspace{5pt}
\item \textbf{Round 2:}\ If chose to transition application file in Round 1, can accept or reject
\vspace{5pt}
\item Assume for now committee transitions and accepts optimal number of files
\vspace{10pt}
\end{itemize}
\item Along with the baseline dynamic model, there are two additional models
\vspace{5pt}
\begin{itemize}
\item Static model makes it possible to compare cutoff probabilities across rounds
\vspace{5pt}
\item Waitlist model adds a third round and accounts for matriculation probabilities
\vspace{10pt}
\end{itemize}
\end{itemize}
\hyperlink{moa}{\beamerskipbutton{Models of Admissions}}
\end{frame}

\begin{frame}[label=ydm1]
\frametitle{Dynamic Model}
\begin{itemize}
\item \textbf{States:}\ $r=$ round, $x=$ objective variables, $z=$ subjective variables, $t=$ year, \\ $\epsilon_r=$ Type 1 extreme value preference shock with scale parameter $\sigma$
\vspace{5pt}
\item \textbf{Controls:}\ $d_1\in{\{\text{Transition},\text{Reject}\}}$, $d_2\in{\{\text{Accept},\text{Reject}\}}$
\vspace{5pt}
\item \textbf{Outcomes:}\ $y\in{\{\text{Success},\text{Failure}\}}$, $y_r^*=$ marginal outcome
\vspace{10pt}
\end{itemize}
\only<2->{\noindent
\textbf{Round 2:}
\begin{equation*}
v_2(x,z,t,\epsilon_2) = \max_{d_2\in\{A,R\}}u_2(x,z,t,d_2)+\epsilon_2(d_2), \: \text{such that:}
\end{equation*}
\begin{equation*}
u_2(x,z,t,d_2) = \underbrace{P(y=S|x,z)\mathds{1}(d_2=A)}_{\text{If accept, get $P(\text{Success})$}}+\underbrace{P(y_2^*=S|t)\mathds{1}(d_2=R)}_{\text{If reject, get $P(\text{Cutoff})$}}
\end{equation*}
}
\hyperlink{ydp}{\beamerskipbutton{Dynamic Parameterization}}
\end{frame}

\addtocounter{framenumber}{-1}
\begin{frame}[label=ydm2]
\frametitle{Dynamic Model}
\textbf{Round 1:}
\begin{equation*}
v_1(x,t,\epsilon_1) = \max_{d_1\in\{T,R\}}u_1(x,t,d_1)+\epsilon_1(d_1), \: \text{such that:}
\end{equation*}
\begin{equation*}
u_1(x,t,d_1) = \underbrace{\mathbb{E}[v_2(x,z,t,\epsilon_2)|x,t]\mathds{1}(d_1=T)}_{\text{If transition, get $\mathbb E[\text{Social Surplus}]$}}+\underbrace{L(y_1^*=S|t)\mathds{1}(d_1=R)}_{\text{If reject, get $L(\text{Cutoff})$}}, \: \text{such that:}
\end{equation*}
\begin{equation*}
\mathbb{E}[v_2(x,z,t,\epsilon_2)|x,t] = \int_{\widetilde z}\sigma\log\Big(e^{P(y=S|x,\widetilde z)/\sigma}+e^{P(y_2^*=S|t)/\sigma}\Big)dF(\widetilde z|x)
\end{equation*}
\only<2->{\noindent
\textbf{Conditional Choice Probabilities:}
\begin{equation*}
P(d_r|x,(z),t) = \frac{\exp(u_r(x,(z),t,d_r)/\sigma)}{\sum_{\widetilde{d_r}\in D_r}\exp(u_r(x,(z),t,\widetilde{d_r})/\sigma)}
\end{equation*}
}
\hyperlink{ydp}{\beamerskipbutton{Dynamic Parameterization}}
\end{frame}

\begin{frame}[label=ysm]
\frametitle{Static Model}
\begin{itemize}
\item \textbf{Goal:}\ Make Round 1 cutoff a probability bounded between 0 and 1
\vspace{5pt}
\end{itemize}
\only<2->{\noindent
\textbf{Round 2:}
\begin{equation*}
v_2(x,z,t,\epsilon_2) = \max_{d_2\in\{A,R\}}u_2(x,z,t,d_2)+\epsilon_2(d_2), \: \text{such that:}
\end{equation*}
\begin{equation*}
u_2(x,z,t,d_2) = \underbrace{P(y=S|x,z)\mathds{1}(d_2=A)}_{\text{If accept, get $P(\text{Success})$}}+\underbrace{P(y_2^*=S|t)\mathds{1}(d_2=R)}_{\text{If reject, get $P(\text{Cutoff})$}}
\end{equation*}
}
\only<3->{\noindent
\textbf{Round 1:}
\begin{equation*}
v_1(x,t,\epsilon_1) = \max_{d_1\in\{T,R\}}u_1(x,t,d_1)+\epsilon_1(d_1), \: \text{such that:}
\end{equation*}
\begin{equation*}
u_1(x,t,d_1) = \underbrace{P(y=S|x)\mathds{1}(d_1=T)}_{\text{If transition, get $P(\text{Success})$}}+\underbrace{P(y_1^*=S|t)\mathds{1}(d_1=R)}_{\text{If reject, get $P(\text{Cutoff})$}}
\end{equation*}
\vspace{2.5pt}
}
\only<4->{\noindent
\begin{itemize}
\item $P(y=S|x)$ estimates inconsistent, but fitted values account for $x$'s effect on $z$
\vspace{5pt}
\end{itemize}
}
\hyperlink{ysp}{\beamerskipbutton{Static Parameterization}}
\end{frame}

\begin{frame}[label=pcwm]
\frametitle{Portfolio Choice with Matriculation}
\begin{itemize}
\item Accept subset of applicants with highest sum of success times matriculation probabilities subject to capacity constraint on expected number of matriculants
\only<1>{
\vspace{10pt}
\item For each applicant $n\in\{1,\dots,N\}$, make decision $d_n\in\{\text{Accept},\text{Reject}\}$ given:
}
\end{itemize}
\begin{equation*}
\max_{(d_1,\dots,d_N)\in\{A,R\}^N} \sum_{n=1}^N\underbrace{P(y_n=S|x_n,z_n)}_{\text{Success probability}}\underbrace{P(m_n=M|x_n,z_n)}_{\text{Matriculation probability}}\underbrace{\mathds1(d_n=A)}_{\text{Acceptance}},
\end{equation*}
\begin{equation*}
\text{such that:} \: \sum_{n=1}^N\underbrace{P(m_n=M|x_n,z_n)}_{\text{Matriculation probability}}\underbrace{\mathds1(d_n=A)}_{\text{Acceptance}}\leq N_M\only<1>{,}
\end{equation*}
\only<1>{\noindent
\begin{itemize}
\item[] where $(x,z)=$ variables, $y\in\{\text{Success},\text{Failure}\}$, $m\in\{\text{Matriculate},\text{Decline}\}$, \\ $N=$ number of applicants, and $N_M=$ maximum number of matriculants
\vspace{10pt}
\end{itemize}
}
\only<2->{\noindent
\begin{itemize}
\item For large $N$, this problem reduces to cutoff rule problem on \textbf{success} probabilities
\vspace{5pt}
\begin{itemize}
\item Likely matriculants more valuable, but also more expensive $\Rightarrow$ effects cancel
\vspace{5pt}
\item \textbf{Upshot:}\ Matriculation probabilities only matter if there are dynamic effects
\vspace{10pt}
\end{itemize}
\end{itemize}
\hyperlink{crwm}{\beamerskipbutton{Cutoff Rule with Matriculation}}
}
\end{frame}

\begin{frame}[label=wme]
\frametitle{Waitlist Memory}
\begin{itemize}
\item Respecify Round 3 cutoff $P(y_3^*=S|t)$, where $t=$ year, as $P(y_3^*=S|\overline{m_{2A}})$
\vspace{5pt}
\begin{itemize}
\item $\overline{m_{2A}}=$ average early accepted applicant's matriculation probability based on $(x,z)$
\vspace{5pt}
\item Enthusiasm variables can affect matriculation independently of success
\vspace{5pt}
\item $P(y_3^*=S|\overline{m_{2A}})$ increases in $\overline{m_{2A}}$ because larger $\overline{m_{2A}}$ means fewer openings
\vspace{10pt}
\end{itemize}
\item What information available at start of Round 2 determines $\overline{m_{2A}}$?
\vspace{5pt}
\begin{itemize}
\item For all transitioned applicants, $\ddot{m_2}$. $\ddot{m_2}$ is defined as $\overline{m_2}^t$ with $m_2$ excluded, where $m_2=$ applicant's own matriculation probability. Within $t$, low $m_2$ implies high $\ddot{m_2}$
\vspace{5pt}
\item Estimate $f_{{\overline{m_{2A}}|\ddot{m_2}}}$ on all transitioned applicants; slope parameter's expected sign is positive provided there is variation in $\overline{m_2}^t$ across years (e.g.\ due to Covid)
\vspace{10pt}
\end{itemize}
\end{itemize}

\begin{frame}[label=wd]
\frametitle{Waitlist Dynamics}
\begin{theorem}[\href{https://jeremyarosen.com/papers/grad_admissions_proofs.pdf}{\underline{Proof}}]
Success probability equal, an applicant's Round 2 acceptance probability is strictly increasing in their matriculation probability.
\end{theorem}
\vspace{10pt}
\begin{itemize}
\item Compared to high $m_2$ applicant, low $m_2$ applicant has high $\ddot{m_2}$, and therefore high $\overline{F}_{\overline{m_{2A}}|\ddot{m_2}}$, $\mathbb{E}[P(\text{Cutoff})_3]$, and $\mathbb{E}[v_3]$. This increases relative payoff of waitlisting them\!
\vspace{10pt}
\item $\mathbb{E}[v_3]$ discounted so payoff from acceptance in Round 3 less than in Round 2
\vspace{5pt}
\begin{itemize}
\item Captures chance of losing applicant to another university between rounds
\vspace{5pt}
\item If committee is more selective in Round 2, discount factor is closer to 1
\vspace{10pt}
\end{itemize}
\item[$\bigstar$] \textbf{Upshot:}\ Accepting applicants who want to matriculate benefits the program
\vspace{10pt}
\end{itemize}

\begin{frame}[label=mle]
\frametitle{Maximum Likelihood Estimation}
\begin{multline*}
\mathcal{L}_N^U\big(\theta\big) = \prod_{n=1}^{N_T}\underbrace{f(z_n|x_n;\theta_F)}_{\text{Density of $z|x$}}\underbrace{P_{d_{1,n}}(\theta_T)P_{d_{2,n}}(\theta_A)^{\mathds{1}(d_{2,n}=A)}(1-P_{d_{2,n}}(\theta_A))^{\mathds{1}(d_{2,n}=R)}}_{\text{Conditional choice probabilities}} \\
\underbrace{P_{y_n}(\theta_S)^{\mathds{1}(y_n=S)}(1-P_{y_n}(\theta_S))^{\mathds{1}(y_n=F)}}_{\text{Outcome probabilities}}\prod_{n=N_T+1}^N\underbrace{f(z_n|x_n;\theta_F)}_{\text{Density of $z|x$}}\underbrace{(1-P_{d_{1,n}}(\theta_T))}_{\text{Conditional choice probability}}
\end{multline*}
\begin{itemize}
\item In unrestricted likelihood $\mathcal{L}^U$, parameterize CCPs with flexible logit specifications
\vspace{5pt}
\item In restricted likelihood $\mathcal{L}^R$, use model CCPs, which depend on structural parameters\!\!
\vspace{5pt}
\item Do success probabilities implied by outcomes align with those implied by CCPs?
\vspace{10pt}
\end{itemize}

\begin{frame}[label=cpi]
\frametitle{Cutoff Probability Identification}
\begin{theorem}[\href{https://jeremyarosen.com/papers/grad_admissions_proofs.pdf}{\underline{Proof}}]
The maximum likelihood cutoff probability estimate is such that the number of acceptances $N_A$ equals the expected $N_A$. Also, for any cutoff probability and $N_A$, \\ the likelihood is highest when the committee accepts the best $N_A$ applicants.
\end{theorem}
\vspace{5pt}
\begin{itemize}
\item Identify each year's cutoffs by how selective committee is that year
\vspace{10pt}
\item While selectivity identifies cutoffs, sorting improves restricted likelihood
\vspace{5pt}
\begin{itemize}
\item If committee engages in sorting, likelihood-ratio test less likely to reject
\vspace{5pt}
\item $\text{LR} = 2\big[\log(\hat{\mathcal L}^U)-\log(\hat{\mathcal L}^R)\big]\sim\chi_Q^2$, such that $Q=\dim(\theta^U)-\dim(\theta^R)$
\vspace{10pt}
\end{itemize}
\end{itemize}

\section{Optimality Tests}
\addtocounter{framenumber}{-1}
\begin{frame}[label=lrdgt]
\frametitle{Likelihood-Ratio and Deviation Gains Tests}
\begin{itemize}
\item \textbf{Likelihood-Ratio:}\ Determines whether model fits data generated by committee's decision rule, but not whether committee can do better by changing decision rule
\vspace{10pt}
\item \textbf{Deviation Gains:}\ Suppose committee adopts model's decision rule. Will average success probability of acceptance/matriculation set increase, and if so, by how much?\!\!\!\!\!\!
\vspace{5pt}
\begin{itemize}
\item Solve model with $\sigma=0$ in CCPs, compute average yearly $P(\text{Success})$ of model's acceptance/matriculation set, and compare to that of actual decision rule's set
\vspace{5pt}
\item $\hat\theta_S^R$ may be biased to rationalize suboptimal behavior, so plug in $\hat\theta_S^U$
\vspace{5pt}
\item \textbf{Example:}\ If committee puts too little weight on \textit{Advanced Math}, then its \\ restricted success probability parameter estimate will be biased down
\vspace{10pt}
\end{itemize}
\end{itemize}
\hyperlink{otn}{\beamerskipbutton{Optimality Tests Notes}}
\end{frame}

\begin{frame}[label=dgi]
\frametitle{Deviation Gains Interpretation}
\begin{itemize}
\item Suppose there are meaningfully large deviation gains. Then committee may have incorrect beliefs about which types of applicants are most likely to succeed
\vspace{5pt}
\begin{itemize}
\item Matriculation rate of model's acceptance set may be less than that of actual decision rule's acceptance set, making estimated gains an upper bound
\vspace{5pt}
\item In waitlist counterfactual, compare success probabilities of matriculation sets by simulating accepted applicants' matriculation decisions after Rounds 2 and 3
\vspace{10pt}
\end{itemize}
\item[$\bigstar$] \textbf{Robustness:}\ Given too few observations for holdout set and infeasibility of \\ large $k$-fold cross-validation due to runtime, implement randomization test
\vspace{5pt}
\begin{itemize}
\item Compute average success probabilities with perturbations of $\hat\theta_S^U$
\vspace{10pt}
\end{itemize}
\end{itemize}
\hyperlink{hdgtw}{\beamerskipbutton{Deviation Gains Test}}
\end{frame}

\begin{itemize}
\item Hard to obtain consistent estimates due to omitted variables
\vspace{5pt}
\begin{itemize}
\item \textbf{Example:}\ Don't observe effort (e.g.\ hours worked). Can bias coefficients upward if predictors correlated with effort and success are independent of effort

\item Existing literature also suffers from this problem, though BEMS \citeyearp{bai2022} parse recommendation letters for ``grindstone terms" (e.g.\ depend*, diligen*)
\vspace{10pt}
\end{itemize}
\item Research questions emphasize prediction over inference
\vspace{5pt}
\begin{itemize}
\item \citet{mullainathan2017} distinguish $\hat\beta$ vs.\ $\hat y$ problems
\vspace{5pt}
\item If GRE predicts success, committee should use it regardless of why
\vspace{5pt}
\item See Reduced-Form Summary for prediction-based methods
\vspace{10pt}
\end{itemize}

\begin{frame}[label=sas]
\frametitle{A1:\ Sample Selection}
\begin{itemize}
\item Determinants of success conditional on applying vs.\ matriculating
\vspace{10pt}
\item BEMS \citeyearp{bai2022} prove that if $P(\text{Success}|\text{Skill})$ is an S-curve, then $\hat\beta$ will be smaller for more selected samples. Consistent with empirical results
\vspace{10pt}
\item If there aren't some types of outcomes data (e.g.\ grades, passing comps) for all applicants, or such data are too hard to obtain, use Heckman correction
\vspace{5pt}
\begin{itemize}
\item Let $y_n^*=x_n\beta+u_n$, $d_n=\mathds{1}(z_n\gamma+v_n>0)$, and $y_n=d_ny_n^*$
\vspace{5pt}
\item \textbf{Step 1:}\ Run probit of $d$ on $z$ and compute inverse Mills $\hat\lambda_n=\frac{\phi(z_n\hat\gamma)}{\Phi(z_n\hat\gamma)}$
\vspace{5pt}
\item \textbf{Step 2:}\ Run OLS, probit, or ordered probit of $y$ on $(x,\hat\lambda)$ to get $\hat\beta$
\vspace{5pt}
\item \textbf{Instruments:}\ Enthusiasm score, average acceptance rate/quality
\vspace{10pt}
\end{itemize}
\end{itemize}

\begin{frame}[label=ioca]
\frametitle{A1:\ Insufficiency of Comparing AMEs}
\begin{itemize}
\item Cannot test optimality by comparing AMEs across regressions. Suppose:
\vspace{5pt}
\begin{itemize}
\item Math skills have large positive effect on success, coding skills smaller positive effect
\vspace{5pt}
\item 25\% of applicants have both skills, 25\% math only, 25\% coding only, 25\% neither
\vspace{5pt}
\item Committee can accept half of applicants, and other half attend different university
\vspace{10pt}
\end{itemize}
\item Optimizing committee accepts half with math and ignores coding
\vspace{5pt}
\begin{itemize}
\item \textbf{Acceptance Regression:}\ $\text{AME}_{\text{Math}}=1$, $\text{AME}_{\text{Coding}}=0$
\vspace{5pt}
\item \textbf{Success Regression:}\ $1>\text{AME}_{\text{Math}}>\text{AME}_{\text{Coding}}>0$
\vspace{5pt}
\item Despite optimality, acceptance and success AMEs not equal across regressions
\vspace{10pt}
\end{itemize}
\item \textbf{Upshot:}\ Relationship between AMEs must be interpreted through a model
\vspace{10pt}
\end{itemize}

\begin{frame}[label=lowt]
\frametitle{A1:\ Logit Wald Test Across Programs}
Assume $\big(Y^E\indep Y^{NE}\big)\big|X$. Let $P\big(Y_n^E=1\big|x_n^E;\theta^E\big) = \frac{\exp(x_{n,0}^E\theta_0^E+x_{n,1}^E\theta_1^E)}{1+\exp(x_{n,0}^E\theta_0^E+x_{n,1}^E\theta_1^E)}$, \\ and $P\big(Y_n^{NE}=1\big|x_n^{NE};\theta^{NE}\big) = \frac{\exp(x_{n,0}^{NE}\theta_0^{NE}+x_{n,1}^{NE}\theta_1^{NE})}{1+\exp(x_{n,0}^{NE}\theta_0^{NE}+x_{n,1}^{NE}\theta_1^{NE})}$. Finally, let $N=N_E+N_{NE}$. Then:
\begin{multline*}
\mathcal{L}_N^U\big(y\big|x;\theta\big) = \prod_{n=1}^{N_E}P\big(Y_n^E=1\big|x_n^E;\theta^E\big)^{Y_n^E}\big[1-P\big(Y_n^E=1\big|x_n^E;\theta^E\big)\big]^{1-Y_n^E}\\
\prod_{n=1}^{N_{NE}}P\big(Y_n^{NE}=1\big|x_n^{NE};\theta^{NE}\big)^{Y_n^{NE}}\big[1-P\big(Y_n^{NE}=1\big|x_n^{NE};\theta^{NE}\big)\big]^{1-Y_n^{NE}}.
\end{multline*}
\begin{itemize}
\item $Y \in \{\mathds{1}(\text{Accept}), \mathds{1}(\text{Attend Program}), \mathds{1}(\text{Complete Program}), \mathds{1}(\text{Good Placement})\}$
\vspace{2.5pt}
\item \textbf{Test:}\ Let $g(\hat\theta)\!=\!\frac{1}{N_E}\!\sum_{n=1}^{N_E}\!\hat P_n^E(1\!-\!\hat P_n^E)\hat\theta_1^E\!-\!\frac{1}{N_{NE}}\!\sum_{n=1}^{N_{NE}}\!\hat P_n^{NE}(1\!-\!\hat P_n^{NE})\hat\theta_1^{NE}$. Then: \\ $\text{W} = g(\hat\theta)'[\nabla_\theta g(\hat\theta)\hat V(\hat\theta)\nabla_\theta g(\hat\theta)'\big]^{-1}g(\hat\theta)\sim\chi_Q^2$. Compare AMEs per \citet{mood2010}
\end{itemize}

\end{frame}

\begin{frame}[label=liwt]
\frametitle{A1:\ Linear Wald Test Across Programs}
Assume $\big(Y^E\indep Y^{NE}\big)\big|X$, i.e.\ $\big(\text{Economics}\indep\text{Non-Economics}\big)\big|\text{Predictors}$. \\
Let $f\big(y_n^E\big|x_n^E;\beta^E,\sigma^2_E\big) = \frac{1}{\sqrt{2\pi\sigma^2_E}}\exp\Big[\frac{-1}{2\sigma^2_E}\big(y_n^E-x_{n,0}^E\beta_0^E-x_{n,1}^E\beta_1^E\big)^2\Big]$, and \\
$f\big(y_n^{NE}\big|x_n^{NE};\beta^{NE},\sigma^2_S\big) = \frac{1}{\sqrt{2\pi\sigma^2_{NE}}}\exp\Big[\frac{-1}{2\sigma^2_{NE}}\big(y_n^{NE}-x_{n,0}^{NE}\beta_0^{NE}-x_{n,1}^{NE}\beta_1^{NE}\big)^2\Big]$. Then:
\begin{equation*}
\mathcal{L}_N^U\big(y\big|x;\beta,\sigma^2\big) = \prod_{n=1}^{N_E}f\big(y_n^E\big|x_n^E;\beta^E,\sigma^2_E\big)\prod_{n=1}^{N_{NE}}f\big(y_n^{NE}\big|x_n^{NE};\beta^{NE},\sigma^2_{NE}\big).

\item \textbf{Test:}\ Let $g(\hat\beta_1) = \hat\beta_1^E-\hat\beta_1^{NE}$. (Can also do likelihood-ratio test.) \\ Then $\text{W} = g(\hat\beta_1)'[\nabla_{\beta,\sigma^2} g(\hat\beta_1)\hat V(\hat\beta,\hat\sigma^2)\nabla_{\beta,\sigma^2} g(\hat\beta_1)'\big]^{-1}g(\hat\beta_1)\sim\chi_Q^2$
\vspace{2.5pt}
\item \textbf{Assumption:}\ Homoskedasticity (i.e.\ $\forall n, \: \sigma^2_n=\sigma^2$)
\vspace{2.5pt}
\end{itemize}

\begin{frame}[label=pctcr]
\frametitle{A2.1:\ Portfolio Choice to Cutoff Rule (BEMS \citeyear{bai2022})}
\textbf{Portfolio Choice Problem:}
\begin{equation*}
\max_{(d_1,\dots,d_N)\in\{A,R\}^N} \sum_{n=1}^NP(y_n=S|x_n,z_n)\mathds1(d_n=A), \: \text{such that:} \: \sum_{n=1}^N\mathds1(d_n=A)\leq N_A
\end{equation*}
\begin{equation*}
\mathcal L(d_1,\dots,d_N;\lambda) = \sum_{n=1}^N\big[P(y_n=S|x_n,z_n)-\lambda\big]\mathds1(d_n=A)+\lambda N_A
\end{equation*}
\textbf{Cutoff Rule Solution:}
\vspace{-7.5pt}
\begin{equation*}
\forall n\in\{1,\dots,N\}, \: d_n(x_n,z_n|\lambda^*)=\begin{cases}
A & \text{if } P(y_n=S|x_n,z_n)>\lambda^* \\
\{A,R\} & \text{if } P(y_n=S|x_n,z_n)=\lambda^* \\
R & \text{if } P(y_n=S|x_n,z_n)<\lambda^*
\end{cases}
\end{equation*}
\textbf{Upshot:}\ $\lambda^*$ is cutoff probability. Denote it hereafter as $P(y^*=S)$
\vspace{2.5pt}
\begin{itemize}
\item $P(y^*=S)\in\big[P(y_{N_A+1}=S|x_{N_A+1},z_{N_A+1}),P(y_{N_A}=S|x_{N_A},z_{N_A})\big]$
\end{itemize}
\hyperlink{mf}{\beamerskipbutton{Model Framework}}
\end{frame}

\begin{frame}[label=moa]
\frametitle{A2.1:\ Models of Admissions}
\begin{enumerate}
\item \textbf{Year-Static Model:}\ Committee ranks applicants based on objective variables, reads files of applicants above Cutoff 1, re-ranks surviving applicants based on all variables, and accepts surviving applicants above Cutoff 2
\vspace{10pt}
\item \textbf{Year-Dynamic Model:}\ Committee forms expectation in Round 1 about applicants who survive to Round 2. Favors applicants with objective variables positively correlated with good subjective variables
\vspace{10pt}
\item \textbf{Waitlist-Hybrid/Dynamic Models:}\ After Round 1, committee accepts or waitlists applicants in Round 2. After accepted applicants matriculate or don't matriculate, committee accepts or rejects waitlisted applicants in Round 3. \\ Favors likely matriculants in Round 2 to increase selectivity in Round 3
\vspace{10pt}
\end{enumerate}

\end{frame}

\begin{frame}[label=cia]
\frametitle{A2.1:\ Conditional Independence Assumption}
\begin{itemize}
\item Should I assume success is independent of admission conditional on predictors?
\vspace{2.5pt}
\begin{itemize}
\item This university aside, if you don't get into anywhere good, you're unlikely to succeed\!
\vspace{2.5pt}
\item Can bias $\hat\theta_S$ upward since without loss of generality, a low GRE is correlated with not getting in anywhere good, which itself prevents you from succeeding
\vspace{5pt}
\end{itemize}
\item So should I replace $P\big(y=S\big|x;\theta_S\big)$ with $P\big(y=S\big|d,x;\theta_S\big)$?
\vspace{2.5pt}
\begin{itemize}
\item \textbf{Problem 1:}\ If I condition $y$ on $d=\mathds{1}$(Accept), I would be ``logistically regressing" $d$ on $y$ on $d$ in restricted likelihood, which will fail
\vspace{2.5pt}
\item \textbf{Problem 2:}\ Because $d$ is a function of $x$, even conditioning $y$ on $x$ and \\ $d=\mathds{1}$(Attend Program) will ``collapse" $\hat\theta_S$ (except for $\hat\theta_S^d$)
\vspace{2.5pt}
\item[$\bigstar$] \textbf{Example:}\ In AKKLP \citeyearp{athey2007}, quantitative GRE predicts course grades but not when Committee Score is included; coefficient even goes negative for Metrics Grade
\vspace{2.5pt}
\item \textbf{Problem 3:}\ Committee doesn't know \textit{ex-ante} whether $d$ will be 1 or 0
\vspace{5pt}
\end{itemize}
\end{itemize}
\hyperlink{mle}{\beamerskipbutton{Maximum Likelihood Estimation}}
\end{frame}

\begin{frame}[label=ec]

\begin{itemize}
\item \textbf{Selection Effects:}\ Track success for all applicants, including non-matriculants
\vspace{5pt}
\begin{itemize}
\item In selected samples, success parameters are biased down (e.g.\ quantitative GRE)
\vspace{10pt}
\end{itemize}
\item \textbf{Potential Outcomes:}\ Track which graduate program all applicants attended, \\ and add expected program ranking conditional on $(x,z)$ as control function
\vspace{5pt}
\begin{itemize}
\item If a variable only predicts success by helping applicant get into elite graduate program, then committee should not use that variable to rank applicants
\vspace{10pt}
\end{itemize}
\item \textbf{Unobservables:}\ I may not observe all information that committee observes
\vspace{5pt}
\begin{itemize}
\item \textbf{Example:}\ Recommendation letter has nuances that algorithms can't detect
\vspace{5pt}
\item Include subjective committee score in $z$ as a proxy for this information
\vspace{5pt}
\end{itemize}
\end{itemize}

\begin{frame}[label=crwm]
\frametitle{A2.1:\ Cutoff Rule with Matriculation}
\textbf{Portfolio Choice Problem:}
\begin{equation*}
\max_{\delta\in\{\delta|\delta: \mathbb R^K\rightarrow[0,1]\}}\int_{(x,z)\in\mathbb R^K}P(y=S|x,z)P(m=M|x,z)\delta(x,z)dF(x,z),
\end{equation*}
\begin{equation*}
\text{such that:} \: \int_{(x,z)\in\mathbb R^K}P(m=M|x,z)\delta(x,z)dF(x,z)\leq N_M
\end{equation*}
\begin{equation*}
\mathcal L(\delta;\lambda) = \int_{(x,z)\in\mathbb R^K}\big[P(y=S|x,z)-\lambda\big]P(m=M|x,z)\delta(x,z)dF(x,z)+\lambda N_M
\end{equation*}
\textbf{Cutoff Rule Solution:}
\vspace{-7.5pt}
\begin{equation*}
\forall (x,z)\in\mathbb R^K, \: \delta(x,z|\lambda^*)=\begin{cases}
1 & \text{if } P(y=S|x,z)>\lambda^* \\
[0,1] & \text{if } P(y=S|x,z)=\lambda^* \\
0 & \text{if } P(y=S|x,z)<\lambda^*
\end{cases}
\end{equation*}
\textbf{Upshot:}\ Point-identified cutoff probability $\lambda^*$ is with respect to success probabilities

\end{frame}

\begin{frame}[label=wm1]
\frametitle{A2.1:\ Waitlist Model 1}
\textbf{Round 3:}
\vspace{-5pt}
\begin{equation*}
v_3(x,z,\overline{m_{2A}},\epsilon_3) = \max_{d_3\in\{A,R\}}u_3(x,z,\overline{m_{2A}},d_3)+\epsilon_3(d_3), \: \text{such that:}
\end{equation*}
\begin{equation*}
u_3(x,z,\overline{m_{2A}},d_3) = \underbrace{P(y=S|x,z)\mathds{1}(d_3=A)}_{\text{If accept, get $P(\text{Success})$}}+\underbrace{P(y_3^*=S|\overline{m_{2A}})\mathds{1}(d_3=R)}_{\text{If reject, get $P(\text{Cutoff})$}}

\end{equation*}
\textbf{Round 2:}
\vspace{-5pt}
\begin{equation*}
v_2(x,z,\ddot{m_2},\epsilon_2) = \max_{d_2\in\{A,W\}}u_2(x,z,\ddot{m_2},d_2)+\epsilon_2(d_2), \: \text{such that:}
\end{equation*}
\begin{equation*}

\begin{theorem}[\href{https://jeremyarosen.com/papers/grad_admissions_proofs.pdf}{\underline{Proof}}]
Let $d^R$ be the model's decision rule, and $d^U$ the actual decision rule. By optimality:
\vspace{-15pt}
\scriptsize
\begin{multline*}
\frac{1}{N_A}\sum_{n=1}^NP\Big(y_n=S\Big|x_n,z_n;\hat\theta_S^U\Big)\mathds{1}\Big\{d_{1,n}\Big[P_{y_n}\Big(\hat\theta^U\Big)\Big]=T,d_{2,n}^R\Big[P_{y_n}\Big(\hat\theta_S^U\Big)\Big]=A\Big\} \\
\geq \frac{1}{N_A}\sum_{n=1}^NP\Big(y_n=S\Big|x_n,z_n;\hat\theta_S^U\Big)\mathds{1}\Big\{d_{1,n}\Big[P_{y_n}\Big(\hat\theta^U\Big)\Big]=T,d_{2,n}^U\Big[P_{y_n}\Big(\hat\theta_A^U\Big)\Big]=A\Big\}
\end{multline*}
\end{theorem}
\normalsize
\begin{itemize}
\item But for perturbation $\widetilde{\theta_{S,t}^U}$ of $\hat\theta_S^U$'s asymptotic distribution, can have:
\vspace{-5pt}
\scriptsize
\begin{multline*}
\frac{1}{N_A}\sum_{n=1}^NP\Big(y_n=S\Big|x_n,z_n;\widetilde{\theta_{S,t}^U}\Big)\mathds{1}\Big\{d_{1,n}\Big[P_{y_n}\Big(\hat\theta_U\Big)\Big]=T,d_{2,n}^R\Big[P_{y_n}\Big(\hat\theta_S^U\Big)\Big]=A\Big\} \\
< \frac{1}{N_A}\sum_{n=1}^NP\Big(y_n=S\Big|x_n,z_n;\widetilde{\theta_{S,t}^U}\Big)\mathds{1}\Big\{d_{1,n}\Big[P_{y_n}\Big(\hat\theta_U\Big)\Big]=T,d_{2,n}^U\Big[P_{y_n}\Big(\hat\theta_A^U\Big)\Big]=A\Big\}
\end{multline*}
\normalsize
\vspace{-10pt}
\item For waitlist models, compare matriculants, not acceptances
\end{itemize}

\begin{frame}[label=mmdd]

\vspace{2.5pt}
\begin{itemize}
\item For example, $a=\mathds{1}$(TheoryField); theory is less common than applied
\vspace{2.5pt}
\item Let $\overline{a}_{-1}=$ percentage of last year's acceptance set that's theory
\vspace{2.5pt}

\begin{theorem}[\href{https://jeremyarosen.com/papers/grad_admissions_proofs.pdf}{\underline{Proof}}]
\small Consider any two applicants, one with $a=1$ and one with $a=0$, but with equal success probabilities and therefore different values of $x$ or $z$. Suppose $P(a'=1)=p>0$. Then for all sufficiently small $p$, the $a=1$ applicant has a strictly greater Round 2 acceptance probability. \normalsize
\end{theorem}
\vspace{2.5pt}
\item Optimal dynamic decision may not equal optimal myopic one. May accept diverse applicant with $P(\text{Success})<L(\text{Cutoff})$ to increase $P(\text{Success})$ of next year's pool
\vspace{2.5pt}
\begin{itemize}
\item Relative to $a=0$, $a=1$ $\Rightarrow$ higher $\overline{F}_{\overline{a}|a}$ $\Rightarrow$ higher $\mathbb{E}[P(\text{Success})']$ and future EV
\end{itemize}
\end{itemize}
